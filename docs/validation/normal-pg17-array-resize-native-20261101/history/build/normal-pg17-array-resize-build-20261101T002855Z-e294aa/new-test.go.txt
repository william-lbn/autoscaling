func normalPG17ArraySQL(t *testing.T, ctx context.Context, sql string) []byte {
	t.Helper()
	cmd := exec.CommandContext(ctx, "/usr/local/bin/psql", "-XqAt", "-v", "ON_ERROR_STOP=1", "-h", "/data", "-p", "55432", "-U", "postgres", "-d", "postgres", "-c", sql)
	cmd.Env = []string{"PATH=/usr/local/bin:/bin", "LC_ALL=C"}
	cmd.SysProcAttr = &syscall.SysProcAttr{Credential: &syscall.Credential{Uid: 1000, Gid: 1000, Groups: []uint32{1000}}}
	output, err := cmd.CombinedOutput()
	if err != nil {
		t.Fatalf("formal PG SQL failed %v: %s\nSQL=%s", err, output, sql)
	}
	return bytes.TrimSpace(output)
}

// 先mincore而不fault；活prefix确已驻留后才只读触原FD并逐页核原A memcg。
// Shrink尾部只做不fault的mincore，不能由观察者重新分配或将RSS当release。
func normalPG17ArrayPages(t *testing.T, original *arrayRegistryOriginal, registration arrayRegistryWireRequest, target, previous, expectedOwner uint64, shrinking bool) []map[string]any {
	t.Helper()
	pagemap, err := os.Open("/proc/self/pagemap")
	if err != nil {
		t.Fatal(err)
	}
	defer pagemap.Close()
	accounting, err := os.Open("/proc/kpagecgroup")
	if err != nil {
		t.Fatal(err)
	}
	defer accounting.Close()
	rows := make([]map[string]any, 0, 4)
	for i, r := range registration.Ranges {
		active := (target*r.Element + 4095) &^ uint64(4095)
		span := active
		if previous > target {
			span = (previous*r.Element + 4095) &^ uint64(4095)
		}
		original.mu.Lock()
		file := original.current.files[i]
		original.mu.Unlock()
		data, err := unix.Mmap(int(file.Fd()), 0, int(span), unix.PROT_READ, unix.MAP_SHARED)
		if err != nil {
			t.Fatal(err)
		}
		vec := make([]byte, span/4096)
		_, _, errno := unix.Syscall(unix.SYS_MINCORE, uintptr(unsafe.Pointer(&data[0])), uintptr(span), uintptr(unsafe.Pointer(&vec[0])))
		runtime.KeepAlive(data)
		runtime.KeepAlive(vec)
		if errno != 0 {
			unix.Munmap(data)
			t.Fatal(errno)
		}
		for j, present := range vec {
			want := uint64(j)*4096 < active
			if want != (present&1 != 0) {
				unix.Munmap(data)
				t.Fatal("actual array residency differs", i, j, want, present, shrinking)
			}
		}
		if err = unix.Madvise(data[:active], unix.MADV_POPULATE_READ); err != nil {
			unix.Munmap(data)
			t.Fatal(err)
		}
		base := uint64(uintptr(unsafe.Pointer(&data[0])))
		samples := make([]map[string]uint64, 0, 2)
		for off := uint64(0); off < active; off += 4096 {
			var raw [8]byte
			if _, err = pagemap.ReadAt(raw[:], int64((base+off)/4096*8)); err != nil {
				t.Fatal(err)
			}
			pte := binary.LittleEndian.Uint64(raw[:])
			pfn := pte & ((1 << 55) - 1)
			if pte>>63 != 1 || pfn == 0 {
				t.Fatal("resident original FD has no observer PTE", i, off)
			}
			if _, err = accounting.ReadAt(raw[:], int64(pfn*8)); err != nil {
				t.Fatal(err)
			}
			owner := binary.LittleEndian.Uint64(raw[:])
			if owner != expectedOwner {
				t.Fatal("active array page is not original A charge", i, off, owner, expectedOwner)
			}
			if off == 0 || off+4096 == active {
				samples = append(samples, map[string]uint64{"offset": off, "pfn": pfn, "memcg_inode": owner})
			}
		}
		runtime.KeepAlive(data)
		if err = unix.Munmap(data); err != nil {
			t.Fatal(err)
		}
		rows = append(rows, map[string]any{"array": i, "active_pages_resident_and_owner_verified": active / 4096, "tail_pages_nonresident": (span - active) / 4096, "expected_A_inode": expectedOwner, "samples": samples})
	}
	return rows
}
