2422: 	 * This has to happen after we've potentially unshared the file
2423: 	 * descriptor table (so that the pidfd doesn't leak into the child
2424: 	 * if the fd table isn't shared).
2425: 	 */
2426: 	if (clone_flags & CLONE_PIDFD) {
2427: 		int flags = (clone_flags & CLONE_THREAD) ? PIDFD_THREAD : 0;
2428: 
2429: 		/* Note that no task has been attached to @pid yet. */
2430: 		retval = __pidfd_prepare(pid, flags, &pidfile);
2431: 		if (retval < 0)
2432: 			goto bad_fork_free_pid;
2433: 		pidfd = retval;
2434: 
2435: 		retval = put_user(pidfd, args->pidfd);
2436: 		if (retval)
2437: 			goto bad_fork_put_pidfd;
2438: 	}
2487: 	/*
2488: 	 * Ensure that the cgroup subsystem policies allow the new process to be
2489: 	 * forked. It should be noted that the new process's css_set can be changed
2490: 	 * between here and cgroup_post_fork() if an organisation operation is in
2491: 	 * progress.
2492: 	 */
2493: 	retval = cgroup_can_fork(p, args);
2494: 	if (retval)
2495: 		goto bad_fork_put_pidfd;

2497: 	/*
2498: 	 * Now that the cgroups are pinned, re-clone the parent cgroup and put
2499: 	 * the new task on the correct runqueue. All this *before* the task
2500: 	 * becomes visible.
2501: 	 *
2502: 	 * This isn't part of ->can_fork() because while the re-cloning is
2503: 	 * cgroup specific, it unconditionally needs to place the task on a
2504: 	 * runqueue.
2505: 	 */

2507: 	if (retval)
2508: 		goto bad_fork_cancel_cgroup;
2615: 	syscall_tracepoint_update(p);
2616: 	write_unlock_irq(&tasklist_lock);
2617: 
2618: 	if (pidfile)
2619: 		fd_install(pidfd, pidfile);
2620: 
2621: 	proc_fork_connector(p);
2622: 	sched_post_fork(p);
2623: 	cgroup_post_fork(p, args);
2624: 	perf_event_fork(p);
2625: 
2626: 	trace_task_newtask(p, clone_flags);
2627: 	uprobe_copy_process(p, clone_flags);
2628: 	user_events_fork(p, clone_flags);
