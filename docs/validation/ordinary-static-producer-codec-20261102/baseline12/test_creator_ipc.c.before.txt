static void drop(void){struct __user_cap_header_struct h={_LINUX_CAPABILITY_VERSION_3,0};struct __user_cap_data_struct c[2]={{0},{0}};if(getuid()==0){if(setgroups(0,NULL)||setresgid(60004,60004,60004)||setresuid(60004,60004,60004))fail("drop60004");}check(getuid()==60004&&getgid()==60004,"actual role uid/gid");if(syscall(SYS_capset,&h,c)||prctl(PR_SET_NO_NEW_PRIVS,1,0,0,0))fail("cap0 NNP");}

static int receive_wait(struct neon_endpoint *e,uint32_t kind,uint64_t gen,uint64_t seq){char body[NEON_IPC_MAX];size_t len;uint64_t actual;if(neon_wait(e->fd,POLLIN,neon_now()+2000))return -ETIMEDOUT;return neon_receive(e,kind,gen,seq,body,&len,&actual);}
static void child_hold(int fd){char b;while(read(fd,&b,1)<0&&errno==EINTR){}close(fd);}

static void right_send(struct neon_endpoint *e){unsigned char packet[40+sizeof(props)]={0};memcpy(packet,"NDSGWI01",8);neon_put32(packet+8,1);neon_put32(packet+12,NEON_DONE);neon_put64(packet+16,1);neon_put64(packet+24,11);neon_put32(packet+32,sizeof(props));memcpy(packet+40,props,sizeof(props));int extra=open("/dev/null",O_RDONLY|O_CLOEXEC);check(extra>=0,"rights source");union{struct cmsghdr align;char raw[CMSG_SPACE(sizeof(int))];}control={0};struct iovec iov={packet,sizeof packet};struct msghdr msg={.msg_iov=&iov,.msg_iovlen=1,.msg_control=control.raw,.msg_controllen=sizeof control.raw};struct cmsghdr *c=CMSG_FIRSTHDR(&msg);c->cmsg_level=SOL_SOCKET;c->cmsg_type=SCM_RIGHTS;c->cmsg_len=CMSG_LEN(sizeof(int));memcpy(CMSG_DATA(c),&extra,sizeof extra);check(sendmsg(e->fd,&msg,MSG_NOSIGNAL)==sizeof(packet),"send malformed Rights");close(extra);}
static void flow(int mode){struct neon_pair pair;check(!neon_pair_new(&pair),"original pair");int finish[2];check(!pipe2(finish,O_CLOEXEC),"hold pipe");pid_t child=fork();check(child>=0,"fork original");if(!child){close(finish[1]);struct neon_endpoint e;neon_pair_child(&pair,&e);check(receive_wait(&e,NEON_EVENT,1,11)==1,"creator first EVENT");if(mode==2)right_send(&e);else check(!neon_send(&e,NEON_DONE,mode==1?2:1,11,props,sizeof(props)),"child DONE");if(mode==0){check(receive_wait(&e,NEON_EVENT,2,12)==1,"creator reuse EVENT2");check(!neon_send(&e,NEON_DONE,2,12,props,sizeof(props)),"same child DONE2");}child_hold(finish[0]);neon_endpoint_close(&e);_exit(0);}close(finish[0]);struct neon_endpoint e;check(!neon_pair_parent(&pair,child,&e),"original child pin");uint64_t cookie=e.cookie,inode=e.peer_inode;check(!neon_send(&e,NEON_EVENT,1,11,props,sizeof(props)),"main first EVENT");int before=countfd();int result=receive_wait(&e,NEON_DONE,1,11);if(mode==0){check(result==1,"original DONE1");check(!neon_send(&e,NEON_EVENT,2,12,props,sizeof(props)),"main EVENT2");check(receive_wait(&e,NEON_DONE,2,12)==1&&e.cookie==cookie&&e.peer_inode==inode,"same worker object/gen2");}else check(result<0,"malformed message accepted");if(mode==2)check(countfd()==before,"received Rights leaked");check(write(finish[1],"X",1)==1,"release original child");close(finish[1]);join(child);neon_endpoint_close(&e);}
static void foreign_actor(void){struct neon_pair pair;check(!neon_pair_new(&pair),"foreign original pair");int finish[2];check(!pipe2(finish,O_CLOEXEC),"foreign hold");pid_t original=fork();check(original>=0,"original fork");if(!original){close(finish[1]);struct neon_endpoint e;neon_pair_child(&pair,&e);child_hold(finish[0]);neon_endpoint_close(&e);_exit(0);}pid_t foreign=fork();check(foreign>=0,"sameUID foreign fork");if(!foreign){struct neon_endpoint e;neon_pair_child(&pair,&e);check(!neon_send(&e,NEON_DONE,1,11,props,sizeof(props)),"sameUID foreign packet send");neon_endpoint_close(&e);_exit(0);}close(finish[0]);struct neon_endpoint e;check(!neon_pair_parent(&pair,original,&e),"pin original not foreign");check(receive_wait(&e,NEON_DONE,1,11)<0,"same UID inherited endpoint impersonated original actor");check(write(finish[1],"X",1)==1,"release original");close(finish[1]);join(original);join(foreign);neon_endpoint_close(&e);}
static void swapped_object(void){struct neon_pair pair;check(!neon_pair_new(&pair),"swap pair");int finish[2];check(!pipe2(finish,O_CLOEXEC),"swap hold");pid_t original=fork();check(original>=0,"swap fork");if(!original){close(finish[1]);struct neon_endpoint e;neon_pair_child(&pair,&e);child_hold(finish[0]);neon_endpoint_close(&e);_exit(0);}close(finish[0]);struct neon_endpoint e;check(!neon_pair_parent(&pair,original,&e),"swap original pin");int old=e.peer_pidfd;e.peer_pidfd=syscall(SYS_pidfd_open,getpid(),0);check(e.peer_pidfd>=0&&neon_endpoint_check(&e)<0,"wrong live pidfd accepted");close(e.peer_pidfd);e.peer_pidfd=old;uint64_t cookie=e.cookie;e.cookie++;check(neon_endpoint_check(&e)<0,"foreign endpoint cookie accepted");e.cookie=cookie;check(!neon_endpoint_check(&e),"restored actual objects rejected");write(finish[1],"X",1);close(finish[1]);join(original);check(neon_endpoint_check(&e)<0,"dead original accepted");neon_endpoint_close(&e);}
